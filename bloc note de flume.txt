import csv

# Chemin vers le fichier CSV et le fichier de sortie .log
csv_file_path = '/home/cloudera/Downloads/supermarket_data/data.csv'
log_file_path = '/home/cloudera/Downloads/supermarket_data/data.log'

# Fonction pour convertir CSV en .log
def convert_csv_to_log(csv_file, log_file):
    try:
        # Ouvrir le fichier CSV pour lecture
        with open(csv_file, mode='r') as csvfile:
            csvreader = csv.reader(csvfile)

# Ouvrir le fichier de log pour écrire
            with open(log_file, mode='w') as logfile:
                # Écrire une ligne d'entête (optionnel)
                logfile.write("Log conversion from CSV started...\n")

                # Lire et écrire chaque ligne du fichier CSV
                for row in csvreader:
                    log_entry = " | ".join(row)  # Joindre les colonnes avec un séparateur
                    logfile.write(log_entry + "\n")
                
                logfile.write("Log conversion from CSV completed...\n")

    print(f"Conversion réussie ! Le fichier log a été créé à : {log_file}")    
   
   except Exception as e:
        print(f"Erreur lors de la conversion : {e}")

# Appeler la fonction pour convertir le CSV en .log
convert_csv_to_log(csv_file_path, log_file_path)

// le contenu de script : 
  
 # Définition de l'agent
agent.sources = source1
agent.sinks = sink1
agent.channels = channel1


# Source : Lecture des logs système
agent.sources.source1.type = exec
agent.sources.source1.command = tail -F /var/log/messages


# Channel : Mémoire tampon entre source et destination
agent.channels.channel1.type = memory
agent.channels.channel1.capacity = 1000
agent.channels.channel1.transactionCapacity = 100

# Sink : Destination des logs (fichier)
agent.sinks.sink1.type = file_roll
agent.sinks.sink1.sink.directory = /home/cloudera/flume-logs
agent.sinks.sink1.sink.rollInterval = 60

# Lier source et sink via le channel
agent.sources.source1.channels = channel1
agent.sinks.sink1.channel = channel1


/*explication */
 Partie1 : ➡️ 
		sources : source1 → collecte les logs.
		sinks : sink1 → envoie les logs à une destination.
		channels : channel1 → sert de mémoire tampon entre la source et la destination
///////////////////////////////////////////////////////////////////////////////////////
Partie2 :

		➡️  Définit la source qui récupère les logs :
		type = exec → La source exécute une commande en ligne de commande.
		command = tail -F /var/log/messages
		tail -F suit dynamiquement un fichier de logs (ici /var/log/messages) et capture toute nouvelle ligne ajoutée.
		C’est utile pour surveiller en temps réel les logs du système Linux.

////////////////////////////////////////////////////////////////////////////////////////
Partie3 :

		➡️ Définit un canal de type "mémoire" pour stocker temporairement les logs :

		type = memory → Stocke les logs en mémoire (plus rapide mais peut perdre des données en cas de crash).
		capacity = 1000 → Capacité maximale du canal : 1000 événements en mémoire.
transactionCapacity = 100 → Nombre maximal d’événements transférés en une transaction.

///////////////////////////////////////////////////////////////////////////////////////////
Partie4 :
               ➡️Définit la destination où les logs seront stockés :

		type = file_roll → Écrit les logs dans un fichier avec rotation automatique.
		directory = /home/cloudera/flume-logs → Emplacement où les fichiers de logs sont enregistrés.
		rollInterval = 60 → Crée un nouveau fichier de logs toutes les 60 secondes.
